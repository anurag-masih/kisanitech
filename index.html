import cv2
import numpy as np

def compute_laser_targets(frame):
    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
    lower_weed = np.array([35, 40, 40])
    upper_weed = np.array([85, 255, 255])
    
    mask = cv2.inRange(hsv, lower_weed, upper_weed)
    contours, _ = cv2.findContours(mask, 
                    cv2.RETR_EXTERNAL, 
                    cv2.CHAIN_APPROX_SIMPLE)
    
    # Return Cartesian (X, Y) coordinates for laser actuation
    return [cv2.moments(c) for c in contours]
